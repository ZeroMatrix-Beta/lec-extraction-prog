Let $x, y \in K$ with $d_X(x, y) < \delta$.
\begin{enumerate}
    \setcounter{enumi}{0} \item Since $K \subseteq \bigcup B_{\delta(x_i)/2}(x_i)$, there exists $i$ such that $x \in B_{\delta(x_i)/2}(x_i)$.
    \setcounter{enumi}{1} \item Then $d_X(x_i, y) \le d_X(x_i, x) + d_X(x, y) < \frac{\delta(x_i)}{2} + \delta \le \frac{\delta(x_i)}{2} + \frac{\delta(x_i)}{2} = \delta(x_i)$.
    \setcounter{enumi}{2} \item Thus both $x$ and $y$ are in $B_{\delta(x_i)}(x_i)$.
    \setcounter{enumi}{3} \item By the choice of $\delta(x_i)$, we have $f(x), f(y) \in B_{\epsilon/2}(f(x_i))$.
    \setcounter{enumi}{4} \item Finally, $d_Y(f(x), f(y)) \le d_Y(f(x), f(x_i)) + d_Y(f(x_i), f(y)) < \frac{\epsilon}{2} + \frac{\epsilon}{2} = \epsilon$.
\end{enumerate}
This shows that $\delta$ works for all $x, y \in K$ simultaneously.
