\begin{nice-box}[Metasatz: Das Deduktionstheorem]
\begin{theorem}[Deduktionstheorem -- DT]
\label[theorem]{thm:deduction-theorem}
Sei $\mathcal{L}$ eine Signatur, $\Phi$ eine Menge von $\mathcal{L}$-Formeln, und seien $\psi, \varphi$ zwei beliebige $\mathcal{L}$-Formeln. Dann gilt:
\begin{equation}
\label{eq:dt-forward}
\Phi \cup \{\psi\} \vdash \varphi \iff \Phi \vdash \psi \to \varphi
\end{equation}
\end{theorem}
